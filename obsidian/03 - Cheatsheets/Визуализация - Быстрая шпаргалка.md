# Визуализация — Быстрая шпаргалка

## Один числовой признак

```python
sns.histplot(data=df, x='feature', bins=30, kde=True)
sns.boxplot(data=df, x='feature')
sns.violinplot(data=df, y='feature')
```

## Категория

```python
sns.countplot(data=df, x='category')
df['category'].value_counts(normalize=True)
```

## Число по группам

```python
sns.boxplot(data=df, x='group', y='value')
sns.violinplot(data=df, x='group', y='value', inner='quartile')

df.groupby('group')['value'].agg(['mean', 'median', 'std', 'count'])
```

## Категория и target

```python
sns.countplot(data=df, x='category', hue='target')

pd.crosstab(
    df['category'],
    df['target'],
    normalize='index',
)
```

## Два числовых признака

```python
sns.scatterplot(
    data=df,
    x='feature_1',
    y='feature_2',
    hue='target',
    alpha=.6,
)
```

## Корреляции

```python
pearson = df.corr(numeric_only=True, method='pearson')
spearman = df.corr(numeric_only=True, method='spearman')

sns.heatmap(pearson, center=0, cmap='coolwarm')
```
